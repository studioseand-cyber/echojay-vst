# limiter_ab_guard (session L, 7 Oct 2026): a STANDALONE tree. No JUCE, no plugin library - the session rule is
# "build only the limiter processor and the offline harness"; linking the guard suite's archive would be a full
# plugin build. The two product headers it reuses (K-weighting, TruePeakInterp) are header-only and JUCE-free.
#   cmake -S tools/limiter_ab_guard -B build-limiter-ab -DCMAKE_BUILD_TYPE=Release && cmake --build build-limiter-ab -j 2
cmake_minimum_required(VERSION 3.22)
project(limiter_ab_guard CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
find_program(EJ_CCACHE ccache)
if(EJ_CCACHE)
    set(CMAKE_CXX_COMPILER_LAUNCHER "${EJ_CCACHE}")
endif()
add_executable(limiter_ab_guard limiter_ab_guard.cpp)
target_compile_options(limiter_ab_guard PRIVATE -Wall -Wextra -O2)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/limiter_v2_render.cpp")
    add_executable(limiter_v2_render limiter_v2_render.cpp)
    target_compile_options(limiter_v2_render PRIVATE -Wall -Wextra -O2)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/limiter_v2_core_test.cpp")
    add_executable(limiter_v2_core_test limiter_v2_core_test.cpp)
    target_compile_options(limiter_v2_core_test PRIVATE -Wall -Wextra -O2)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/limiter_legacy_render.cpp")
    add_executable(limiter_legacy_render limiter_legacy_render.cpp)
    target_compile_options(limiter_legacy_render PRIVATE -Wall -Wextra -O2)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/blind_pack.cpp")
    add_executable(blind_pack blind_pack.cpp)
    target_compile_options(blind_pack PRIVATE -Wall -Wextra -O2)
endif()
enable_testing()
add_test(NAME limiter_ab_guard_selftest COMMAND limiter_ab_guard selftest)
if(TARGET limiter_v2_core_test)
    add_test(NAME limiter_v2_core_test COMMAND limiter_v2_core_test)
endif()
